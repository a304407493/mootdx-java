## 一、功能概述

持仓笔记功能是一个完整的股票持仓管理系统，支持：
- 多平台管理（支持多个券商账户）
- 持仓记录管理（买入、卖出、费用计算）
- 实时盈亏计算（对接行情数据）
- 交易笔记记录

    market_value FLOAT COMMENT '市值',
    profit_amount FLOAT DEFAULT 0 COMMENT '盈亏金额',
    profit_rate FLOAT DEFAULT 0 COMMENT '盈亏率(%)',

### 3.3 费用计算工具 (utils/fee_calculator.py)

```python
import math
from typing import Dict

class FeeCalculator:
    """股票交易费用计算器"""
    
    @staticmethod
    def calculate_buy_fees(
        amount: float,
        quantity: int,
        commission_rate: float = 0.00025,
        min_commission: float = 5.0,
        transfer_fee_rate: float = 0.00002
    ) -> Dict[str, float]:
        """计算买入费用"""
        # 佣金 = max(成交金额 * 佣金率, 最低佣金)
        commission = max(amount * commission_rate, min_commission)
        
        # 过户费 = 成交金额 * 过户费率 (沪市收取，深市为0)
        transfer_fee = amount * transfer_fee_rate
        
        # 总费用
        total = commission + transfer_fee
        
        return {
            'commission': round(commission, 2),
            'transfer_fee': round(transfer_fee, 2),
            'total': round(total, 2)
        }
    
    @staticmethod
    def calculate_sell_fees(
        amount: float,
        quantity: int,
        commission_rate: float = 0.00025,
        min_commission: float = 5.0,
        stamp_tax_rate: float = 0.001,
        transfer_fee_rate: float = 0.00002
    ) -> Dict[str, float]:
        """计算卖出费用"""
        # 佣金
        commission = max(amount * commission_rate, min_commission)
        
        # 印花税 = 成交金额 * 印花税率 (卖出时收取)
        stamp_tax = amount * stamp_tax_rate
        
        # 过户费
        transfer_fee = amount * transfer_fee_rate
        
        # 总费用
        total = commission + stamp_tax + transfer_fee
        
        return {
            'commission': round(commission, 2),
            'stamp_tax': round(stamp_tax, 2),
            'transfer_fee': round(transfer_fee, 2),
            'total': round(total, 2)
        }
    
    @staticmethod
    def calculate_financing_interest(
        financing_amount: float,
        days: int,
        annual_rate: float = 0.08
    ) -> float:
        """计算融资利息"""
        # 利息 = 融资金额 * 年利率 * 天数 / 365
        interest = financing_amount * annual_rate * days / 365
        return round(interest, 2)
```

### 3.4 业务服务层 (services/stock_note_service.py)

```python
from typing import List, Optional, Dict, Any
from datetime import date, datetime
from sqlalchemy.orm import Session
from sqlalchemy import func
import logging

from app.models.stock_note import TradePlatform, StockPosition
from app.utils.fee_calculator import FeeCalculator

logger = logging.getLogger(__name__)

class StockNoteService:
    """持仓笔记服务"""
    
    def __init__(self, db: Session):
        self.db = db
        self.fee_calculator = FeeCalculator()
    
    # ==================== 平台管理 ====================
    
    def create_platform(self, platform_data: Dict[str, Any]) -> TradePlatform:
        """创建交易平台"""
        platform = TradePlatform(**platform_data)
        self.db.add(platform)
        self.db.commit()
        self.db.refresh(platform)
        return platform
    
    def get_platforms(self, active_only: bool = True) -> List[TradePlatform]:
        """获取平台列表"""
        query = self.db.query(TradePlatform)
        if active_only:
            query = query.filter(TradePlatform.is_active == True)
        return query.order_by(TradePlatform.sort_order).all()
    
    def get_platform_by_id(self, platform_id: int) -> Optional[TradePlatform]:
        """根据ID获取平台"""
        return self.db.query(TradePlatform).filter(TradePlatform.id == platform_id).first()
    
    # ==================== 持仓管理 ====================
    
    def create_position(self, position_data: Dict[str, Any]) -> StockPosition:
        """创建持仓记录"""
        # 获取平台信息用于计算费用
        platform = self.get_platform_by_id(position_data['platform_id'])
        if not platform:
            raise ValueError(f"平台不存在")
        
        # 计算买入金额
        buy_price = position_data['buy_price']
        quantity = position_data['buy_quantity']
        buy_amount = buy_price * quantity
        
        # 计算买入费用
        buy_fees = self.fee_calculator.calculate_buy_fees(
            amount=buy_amount,
            quantity=quantity,
            commission_rate=platform.buy_commission_rate,
            min_commission=platform.buy_min_commission,
            transfer_fee_rate=platform.transfer_fee_rate
        )
        
        # 计算融资利息
        financing_interest = 0
        if position_data.get('is_financing') and position_data.get('financing_amount', 0) > 0:
            financing_interest = self.fee_calculator.calculate_financing_interest(
                financing_amount=position_data['financing_amount'],
                days=position_data.get('financing_days', 0),
                annual_rate=platform.financing_rate
            )
        
        # 构建持仓数据
        position_dict = {
            **position_data,
            'buy_amount': buy_amount,
            'current_quantity': quantity,
            'avg_buy_price': buy_price,
            'buy_commission': buy_fees['commission'],
            'buy_transfer_fee': buy_fees['transfer_fee'],
            'financing_interest': financing_interest,
            'total_fee': buy_fees['total'],
            'total_cost': buy_amount + buy_fees['total'] + financing_interest,
            'status': 'holding'
        }
        
        position = StockPosition(**position_dict)
        self.db.add(position)
        self.db.commit()
        self.db.refresh(position)
        return position
    
    def get_positions(
        self,
        platform_id: Optional[int] = None,
        status: Optional[str] = None,
        code: Optional[str] = None
    ) -> List[StockPosition]:
        """获取持仓列表"""
        query = self.db.query(StockPosition)
        
        if platform_id:
            query = query.filter(StockPosition.platform_id == platform_id)
        if status:
            query = query.filter(StockPosition.status == status)
        if code:
            query = query.filter(StockPosition.code == code)
        
        return query.order_by(StockPosition.buy_date.desc()).all()
    
    def update_position_profit(self, position_id: int, current_price: float) -> Optional[StockPosition]:
        """更新持仓盈亏"""
        position = self.db.query(StockPosition).filter(StockPosition.id == position_id).first()
        if not position or position.status != 'holding':
            return None
        
        # 计算当前市值和盈亏
        market_value = current_price * position.current_quantity
        profit_amount = market_value - position.total_cost
        profit_rate = (profit_amount / position.total_cost * 100) if position.total_cost > 0 else 0
        
        # 更新数据
        position.current_price = current_price
        position.market_value = market_value
        position.profit_amount = round(profit_amount, 2)
        position.profit_rate = round(profit_rate, 2)
        position.updated_at = datetime.now()
        
        self.db.commit()
        self.db.refresh(position)
        return position
    
    def sell_position(
        self,
        position_id: int,
        sell_price: float,
        sell_quantity: Optional[int] = None,
        sell_date: Optional[date] = None
    ) -> Optional[StockPosition]:
        """卖出持仓"""
        position = self.db.query(StockPosition).filter(StockPosition.id == position_id).first()
        if not position:
            return None
        
        platform = self.get_platform_by_id(position.platform_id)
        if not platform:
            return None
        
        # 确定卖出数量
        if sell_quantity is None:
            sell_quantity = position.current_quantity
        
        if sell_quantity > position.current_quantity:
            raise ValueError("卖出数量不能大于持仓数量")
        
        # 计算卖出金额和费用
        sell_amount = sell_price * sell_quantity
        sell_fees = self.fee_calculator.calculate_sell_fees(
            amount=sell_amount,
            quantity=sell_quantity,
            commission_rate=platform.sell_commission_rate,
            min_commission=platform.sell_min_commission,
            stamp_tax_rate=platform.stamp_tax_rate,
            transfer_fee_rate=platform.transfer_fee_rate
        )
        
        # 更新持仓信息
        position.sell_price = sell_price
        position.sell_amount = sell_amount
        position.sell_date = sell_date or date.today()
        position.sell_commission = sell_fees['commission']
        position.sell_stamp_tax = sell_fees['stamp_tax']
        position.sell_transfer_fee = sell_fees['transfer_fee']
        
        # 更新持仓数量
        position.sold_quantity = (position.sold_quantity or 0) + sell_quantity
        position.current_quantity = position.buy_quantity - position.sold_quantity
        
        # 更新状态
        if position.current_quantity <= 0:
            position.status = 'sold'
        else:
            position.status = 'partial'
        
        position.updated_at = datetime.now()
        self.db.commit()
        self.db.refresh(position)
        return position
    
    def get_all_summary(self) -> Dict[str, Any]:
        """获取所有平台汇总统计"""
        # 总体统计
        total_stats = self.db.query(
            func.sum(StockPosition.total_cost).label('total_cost'),
            func.sum(StockPosition.market_value).label('total_market_value'),
            func.sum(StockPosition.profit_amount).label('total_profit'),
            func.count(StockPosition.id).label('position_count')
        ).filter(StockPosition.status.in_(['holding', 'partial'])).first()
        
        total_cost = total_stats.total_cost or 0
        total_market_value = total_stats.total_market_value or 0
        total_profit = total_stats.total_profit or 0
        total_profit_rate = (total_profit / total_cost * 100) if total_cost > 0 else 0
        
        return {
            'total': {
                'total_cost': round(total_cost, 2),
                'total_market_value': round(total_market_value, 2),
                'total_profit': round(total_profit, 2),
                'total_profit_rate': round(total_profit_rate, 2),
                'position_count': total_stats.position_count or 0
            },
            'platforms': []
        }
```

## 五、关键交互流程

### 5.1 添加持仓流程

1. 用户点击"添加持仓"按钮
2. 弹出对话框，用户选择交易平台
3. 用户输入股票代码或名称进行搜索（支持模糊搜索）
4. 选择股票后，股票代码、市场、名称自动填充
5. 用户输入买入价格、数量、日期
6. 可选：设置止损价、目标价、交易策略、标签、笔记
7. 点击确定，后端自动计算买入费用和总成本
8. 刷新列表显示新持仓

### 5.2 卖出持仓流程

1. 用户点击持仓行的"卖出"按钮
2. 弹出卖出对话框，显示当前持仓信息
3. 用户输入卖出价格、数量（可选择全部卖出）
4. 点击确定，后端自动计算卖出费用（佣金、印花税、过户费）
5. 更新持仓状态（持仓中/部分卖出/已清仓）
6. 刷新列表显示更新后的数据

### 5.3 平台管理流程

1. 用户点击"平台设置"按钮
2. 显示平台列表，包含各平台的费率配置
3. 点击"添加平台"，弹出平台配置对话框
4. 输入平台名称、账户类型、各项费率
5. 点击确定保存平台配置
6. 添加持仓时可以选择已配置的平台

## 六、费用计算规则

### 6.1 买入费用

| 费用项 | 计算公式 | 默认值 |
|--------|----------|--------|
| 佣金 | max(成交金额 × 佣金率, 最低佣金) | 万2.5，最低5元 |
| 过户费 | 成交金额 × 过户费率 | 十万分之2 |
| **总费用** | 佣金 + 过户费 | - |

### 6.2 卖出费用

| 费用项 | 计算公式 | 默认值 |
|--------|----------|--------|
| 佣金 | max(成交金额 × 佣金率, 最低佣金) | 万2.5，最低5元 |
| 印花税 | 成交金额 × 印花税率 | 千1 |
| 过户费 | 成交金额 × 过户费率 | 十万分之2 |
| **总费用** | 佣金 + 印花税 + 过户费 | - |

### 6.3 融资利息

| 费用项 | 计算公式 | 默认值 |
|--------|----------|--------|
| 融资利息 | 融资金额 × 年利率 × 天数 / 365 | 年利率8% |

## 七、数据字段说明

### 7.1 持仓状态

| 状态 | 说明 | 触发条件 |
|------|------|----------|
| holding | 持仓中 | 买入后未卖出 |
| partial | 部分卖出 | 卖出部分持仓 |
| sold | 已清仓 | 全部卖出 |

### 7.2 关键字段计算

| 字段 | 计算公式 |
|------|----------|
| buy_amount | buy_price × buy_quantity |
| total_cost | buy_amount + total_fee + financing_interest |
| market_value | current_price × current_quantity |
| profit_amount | market_value - total_cost |
| profit_rate | (profit_amount / total_cost) × 100% |
