# 数据库管理功能完整实现与复刻指南

## 一、功能概述

数据库管理模块提供以下核心功能：
1. **数据表列表展示** - 展示所有数据库表，支持分类筛选
2. **表数据浏览** - 分页查看表数据，支持字段名/备注切换显示
3. **表结构查看** - 查看表的列信息、索引、主键
4. **SQL执行器** - 执行SELECT/INSERT/UPDATE/DELETE语句
5. **SQL模板** - 提供常用SQL模板快速生成
6. **执行历史** - 记录SQL执行历史
7. **【新增】多SQL同时执行** - 支持批量执行多条SQL
8. **【新增】SQL自动提示** - 智能SQL代码补全

```javascript
/**
 * 加载所有表信息
 * GET /api/db-manager/tables
 */
const loadTables = async () => {
  tablesLoading.value = true
  try {
    const response = await axios.get('/api/db-manager/tables')
    tables.value = response.data.tables || []
    categories.value = response.data.categories || []
  } catch (error) {
    console.error('加载表列表失败:', error)
    ElMessage.error('加载表列表失败')
  } finally {
    tablesLoading.value = false
  }
}

/**
 * 选择表
 */
const selectTable = (tableName) => {
  selectedTable.value = tableName
  currentPage.value = 1
  activeTab.value = 'data'
  loadTableData()
  loadTableStructure()
}

/**
 * 加载表数据
 * GET /api/db-manager/tables/{table_name}/data?page={page}&page_size={page_size}
 */
const loadTableData = async () => {
  if (!selectedTable.value) return
  
  dataLoading.value = true
  try {
    const response = await axios.get(`/api/db-manager/tables/${selectedTable.value}/data`, {
      params: {
        page: currentPage.value,
        page_size: pageSize.value
      }
    })
    
    tableData.value = response.data.data || []
    tableColumns.value = response.data.columns || []
    totalRecords.value = response.data.total || 0
  } catch (error) {
    console.error('加载表数据失败:', error)
    ElMessage.error('加载表数据失败')
  } finally {
    dataLoading.value = false
  }
}

/**
 * 加载表结构
 * GET /api/db-manager/tables/{table_name}/structure
 */
const loadTableStructure = async () => {
  if (!selectedTable.value) return
  
  try {
    const response = await axios.get(`/api/db-manager/tables/${selectedTable.value}/structure`)
    structureData.value = response.data
  } catch (error) {
    console.error('加载表结构失败:', error)
    ElMessage.error('加载表结构失败')
  }
}

/**
 * 执行SQL
 * POST /api/db-manager/execute
 */
const executeSql = async () => {
  if (!sqlInput.value.trim()) {
    ElMessage.warning('请输入SQL语句')
    return
  }
  
  sqlExecuting.value = true
  try {
    const response = await axios.post('/api/db-manager/execute', {
      sql: sqlInput.value.trim()
    })
    sqlResult.value = response.data
    
    if (response.data.success) {
      ElMessage.success(response.data.message)
      loadSqlHistory()
    } else {
      ElMessage.error(response.data.message)
    }
  } catch (error) {
    const errorMsg = error.response?.data?.detail || error.message
    sqlResult.value = {
      success: false,
      message: errorMsg,
      execution_time: 0,
      sql_type: null
    }
    ElMessage.error(`SQL执行失败: ${errorMsg}`)
  } finally {
    sqlExecuting.value = false
  }
}

/**
 * 加载SQL历史记录
 * GET /api/db-manager/sql-history?limit=20
 */
const loadSqlHistory = async () => {
  try {
    const response = await axios.get('/api/db-manager/sql-history', {
      params: { limit: 20 }
    })
    sqlHistory.value = response.data.history || []
  } catch (error) {
    console.error('加载SQL历史失败:', error)
  }
}

/**
 * 加载SQL模板
 * GET /api/db-manager/sql-templates
 */
const loadSqlTemplates = async () => {
  try {
    const response = await axios.get('/api/db-manager/sql-templates')
    sqlTemplates.value = response.data.templates || []
    templateCategories.value = response.data.categories || []
  } catch (error) {
    console.error('加载SQL模板失败:', error)
  }
}

/**
 * 应用SQL模板
 * POST /api/db-manager/sql-templates/apply
 */
const applyTemplate = async (template) => {
  try {
    const response = await axios.post('/api/db-manager/sql-templates/apply', {
      template_id: template.id,
      table_name: selectedTable.value
    })
    
    // 插入模板SQL到编辑器
    if (sqlInput.value.trim()) {
      sqlInput.value += '\n\n' + response.data.sql
    } else {
      sqlInput.value = response.data.sql
    }
  } catch (error) {
    const errorMsg = error.response?.data?.detail || error.message
    ElMessage.error(`应用模板失败: ${errorMsg}`)
  }
}
```

```python
from pydantic import BaseModel, Field
from typing import List, Optional, Dict, Any
from datetime import datetime

class TableCategory(BaseModel):
    """表分类模型"""
    name: str = Field(..., description="分类名称")
    description: str = Field(..., description="分类描述")

class TableInfo(BaseModel):
    """表信息模型"""
    name: str = Field(..., description="表名")
    comment: Optional[str] = Field(None, description="表注释")
    row_count: Optional[int] = Field(None, description="行数")
    columns_count: int = Field(..., description="列数")
    category: TableCategory = Field(..., description="业务分类")

class ColumnInfo(BaseModel):
    """列信息模型"""
    name: str = Field(..., description="字段名")
    type: str = Field(..., description="数据类型")
    nullable: bool = Field(..., description="是否可为空")
    default: Optional[str] = Field(None, description="默认值")
    comment: Optional[str] = Field(None, description="字段注释")
    is_primary_key: bool = Field(False, description="是否主键")

class IndexInfo(BaseModel):
    """索引信息模型"""
    name: str = Field(..., description="索引名")
    columns: List[str] = Field(..., description="索引列")
    unique: bool = Field(..., description="是否唯一索引")

class TableStructure(BaseModel):
    """表结构模型"""
    name: str = Field(..., description="表名")
    comment: Optional[str] = Field(None, description="表注释")
    columns: List[ColumnInfo] = Field(..., description="列信息")
    indexes: List[IndexInfo] = Field(..., description="索引信息")
    primary_keys: List[str] = Field(..., description="主键列")

class TableDataResponse(BaseModel):
    """表数据响应"""
    table_name: str = Field(..., description="表名")
    columns: List[Dict[str, Any]] = Field(..., description="列定义")
    data: List[Dict[str, Any]] = Field(..., description="数据")
    total: int = Field(..., description="总条数")
    page: int = Field(..., description="当前页")
    page_size: int = Field(..., description="每页条数")
    total_pages: int = Field(..., description="总页数")

class SQLExecuteRequest(BaseModel):
    """SQL执行请求"""
    sql: str = Field(..., description="SQL语句", min_length=1, max_length=5000)
    params: Optional[Dict[str, Any]] = Field(None, description="SQL参数")

class SQLExecuteResult(BaseModel):
    """SQL执行结果"""
```

```python
_sql_history: List[SQLHistoryItem] = []

def get_sql_type(sql: str) -> str:
    """获取SQL类型"""
    sql_upper = sql.strip().upper()
    if sql_upper.startswith("SELECT"):
        return "SELECT"
    elif sql_upper.startswith("INSERT"):
        return "INSERT"
    elif sql_upper.startswith("UPDATE"):
        return "UPDATE"
    elif sql_upper.startswith("DELETE"):
        return "DELETE"
    else:
        return "OTHER"

def validate_sql(sql: str) -> tuple[bool, str]:
    """验证SQL语句安全性"""
    # 禁止危险操作
    dangerous_keywords = ["DROP", "TRUNCATE", "ALTER", "CREATE", "GRANT", "REVOKE"]
    sql_upper = sql.upper()
    
    for keyword in dangerous_keywords:
        if keyword in sql_upper:
            return False, f"禁止执行包含 {keyword} 的SQL语句"
    
    # 只允许特定的SQL类型
    allowed_prefixes = ["SELECT", "INSERT", "UPDATE", "DELETE", "WITH", "EXPLAIN"]
    has_allowed_prefix = any(sql_upper.strip().startswith(prefix) for prefix in allowed_prefixes)
    
    if not has_allowed_prefix:
        return False, "只允许执行 SELECT/INSERT/UPDATE/DELETE/WITH/EXPLAIN 类型的SQL语句"
    
    return True, ""

            }
            columns_def.append(col_def)

        count_result = db.execute(text(f"SELECT COUNT(*) FROM {table_name}"))

        offset = (page - 1) * page_size
        column_names = [col['name'] for col in columns_info]
        columns_str = ", ".join([f'"{name}"' for name in column_names])
        
        # 构建排序
        order_clause = ""
        if sort_column and sort_column in column_names:
            order_direction = "DESC" if sort_order.lower() == "desc" else "ASC"
            order_clause = f'ORDER BY "{sort_column}" {order_direction}'
        
        # 构建分页查询
        query = f"""
            SELECT {columns_str} 
            FROM {table_name} 
            {order_clause}
            LIMIT {page_size} OFFSET {offset}
        """
        
        result = db.execute(text(query))

            table_name=table_name,
            columns=columns_def,
            data=data,
            total=total,
            page=page,
            page_size=page_size,
            total_pages=total_pages

        raise

@router.post("/execute", response_model=SQLExecuteResult)
async def execute_sql(request: SQLExecuteRequest, db: Session = Depends(get_db)):
    """执行SQL语句"""
    sql = request.sql.strip()
    sql_type = get_sql_type(sql)
    
    # 验证SQL安全性
    is_valid, error_msg = validate_sql(sql)
    if not is_valid:
        # 记录历史
        history_item = SQLHistoryItem(
            id=str(datetime.now().timestamp()),
            sql=sql,
            sql_type=sql_type,
            executed_at=datetime.now(),
            success=False,
            row_count=None,
            execution_time=0
        )
        _sql_history.insert(0, history_item)
        if len(_sql_history) > 100:
            _sql_history.pop()
        
        raise HTTPException(status_code=400, detail=error_msg)
    
    start_time = datetime.now()
    
    try:
        # 执行SQL
        result = db.execute(text(sql), request.params or {})
        
        # 计算执行时间
        execution_time = (datetime.now() - start_time).total_seconds() * 1000
        
        response_data = {
            "success": True,
            "message": "执行成功",
            "execution_time": round(execution_time, 2),
            "sql_type": sql_type
        }
        
        if sql_type == "SELECT" or sql_type == "WITH":
            # 查询操作，返回数据
            rows = result.fetchall()
            columns = result.keys()
            
            # 转换数据
            data = []
            for row in rows:
                row_dict = {}
                for i, col in enumerate(columns):
                    value = row[i]
                    if value is None:
                        row_dict[col] = None
                    elif isinstance(value, (int, float, str, bool)):
                        row_dict[col] = value
                    else:
                        row_dict[col] = str(value)
                data.append(row_dict)
            
            # 构建列信息
            column_info = [{"name": col, "type": "unknown"} for col in columns]
            
            response_data["row_count"] = len(data)
            response_data["data"] = data
            response_data["columns"] = column_info
        else:
            # DML操作
            row_count = result.rowcount
            db.commit()
            
            response_data["row_count"] = row_count
            response_data["message"] = f"执行成功，影响 {row_count} 行"
        
        # 记录历史
        history_item = SQLHistoryItem(
            id=str(datetime.now().timestamp()),
            sql=sql,
            sql_type=sql_type,
            executed_at=datetime.now(),
            success=True,
            row_count=response_data.get("row_count"),
            execution_time=round(execution_time, 2)
        )
        _sql_history.insert(0, history_item)
        if len(_sql_history) > 100:
            _sql_history.pop()
        
        return SQLExecuteResult(**response_data)
        
    except Exception as e:
        db.rollback()
        execution_time = (datetime.now() - start_time).total_seconds() * 1000
        
        # 记录历史
        history_item = SQLHistoryItem(
            id=str(datetime.now().timestamp()),
            sql=sql,
            sql_type=sql_type,
            executed_at=datetime.now(),
            success=False,
            row_count=None,
            execution_time=round(execution_time, 2)
        )
        _sql_history.insert(0, history_item)
        if len(_sql_history) > 100:
            _sql_history.pop()
        
        raise HTTPException(status_code=500, detail=f"SQL执行失败: {str(e)}")
```

```javascript
const executionMode = ref('single') // single|batch|transaction
const batchResults = ref([])
const activeResults = ref([0])

/** 解析SQL语句 */
const sqlStatements = computed(() => {
  if (!sqlInput.value.trim()) return []
  
  // 按分号分割，但忽略引号内的分号
  const statements = []
  let current = ''
  let inString = false
  let stringChar = ''
  
  for (let i = 0; i < sqlInput.value.length; i++) {
    const char = sqlInput.value[i]
    const nextChar = sqlInput.value[i + 1]
    
    // 处理字符串
    if (!inString && (char === "'" || char === '"')) {
      inString = true
      stringChar = char
    } else if (inString && char === stringChar) {
      // 检查是否是转义
      if (nextChar === stringChar) {
        current += char
        i++ // 跳过下一个引号
      } else {
        inString = false
        stringChar = ''
      }
    }
    
    // 处理语句分隔
    if (char === ';' && !inString) {
      const trimmed = current.trim()
      if (trimmed) {
        statements.push(trimmed)
      }
      current = ''
    } else {
      current += char
    }
  }
  
  // 处理最后一条（没有分号结尾）
  const trimmed = current.trim()
  if (trimmed) {
    statements.push(trimmed)
  }
  
  return statements
})

/** SQL语法高亮 */

  let sql = sqlInput.value
    .replace(/&/g, '&amp;')

    .replace(/>/g, '&gt;')

  ]
  
  keywords.forEach(kw => {
    const regex = new RegExp(`\\b${kw}\\b`, 'gi')
    sql = sql.replace(regex, `<span class="sql-keyword">${kw}</span>`)
  })
  
  // 字符串高亮
  sql = sql.replace(/'[^']*'/g, '<span class="sql-string">$&</span>')
  sql = sql.replace(/"[^"]*"/g, '<span class="sql-string">$&</span>')

  sql = sql.replace(/\b\d+\b/g, '<span class="sql-number">$&</span>')
  
  // 注释高亮
  sql = sql.replace(/--[^\n]*/g, '<span class="sql-comment">$&</span>')
  sql = sql.replace(/\/\*[\s\S]*?\*\//g, '<span class="sql-comment">$&</span>')
  
  return sql
})

  if (sqlStatements.value.length === 0) {
    ElMessage.warning('请输入SQL语句')
    return

  batchResults.value = []

    })
    
    batchResults.value = response.data.results
    
    const successCount = batchResults.value.filter(r => r.success).length

  } finally {
    sqlExecuting.value = false
  }
}
```

```python
@router.post("/execute-batch", response_model=BatchSQLExecuteResponse)
async def execute_batch_sql(
    request: BatchSQLExecuteRequest,
    db: Session = Depends(get_db)
):
    """
    批量执行SQL语句
    
    执行模式:
    - single: 逐条执行，每条独立提交，失败不影响其他语句
    - batch: 逐条执行，全部成功后统一提交，失败则回滚
    - transaction: 作为事务执行，任一失败则全部回滚
    """
    start_time = datetime.now()
    results = []
    
    # 验证所有SQL
    for i, sql in enumerate(request.statements):
        sql = sql.strip()
        if not sql:
            continue
            
        is_valid, error_msg = validate_sql(sql)
        if not is_valid:
            results.append(BatchSQLResult(
                index=i,
                sql=sql[:100] + "..." if len(sql) > 100 else sql,
                sql_type=get_sql_type(sql),
                success=False,
                message=f"验证失败: {error_msg}",
                execution_time=0,
                error=error_msg
            ))
    
    # 如果有验证失败的，直接返回
    if any(not r.success for r in results):
        total_time = (datetime.now() - start_time).total_seconds() * 1000
        return BatchSQLExecuteResponse(
            results=results,
            total_count=len(request.statements),
            success_count=0,
            fail_count=len(results),
            total_execution_time=round(total_time, 2),
            mode=request.mode
        )
    
    results = []
    
    try:
        if request.mode == "transaction":
            # 事务模式：全部作为一个事务
            results = await _execute_in_transaction(db, request.statements)
        elif request.mode == "batch":
            # 批处理模式：逐条执行，全部成功后提交
            results = await _execute_in_batch(db, request.statements)
        else:
            # 单条模式：逐条执行，每条独立提交
            results = await _execute_single_mode(db, request.statements)
        
        total_time = (datetime.now() - start_time).total_seconds() * 1000
        success_count = sum(1 for r in results if r.success)
        
        return BatchSQLExecuteResponse(
            results=results,
            total_count=len(request.statements),
            success_count=success_count,
            fail_count=len(results) - success_count,
            total_execution_time=round(total_time, 2),
            mode=request.mode
        )
        
    except Exception as e:
        db.rollback()
        logger.error(f"[数据库管理] 批量执行失败: {e}")
        raise HTTPException(status_code=500, detail=f"批量执行失败: {str(e)}")


async def _execute_single_mode(db: Session, statements: List[str]) -> List[BatchSQLResult]:
    """单条模式执行 - 每条独立提交"""
    results = []
    
    for i, sql in enumerate(statements):
        sql = sql.strip()
        if not sql:
            continue
            
        sql_type = get_sql_type(sql)
        start = datetime.now()
        
        try:
            result = db.execute(text(sql))
            execution_time = (datetime.now() - start).total_seconds() * 1000
            
            if sql_type == "SELECT" or sql_type == "WITH":
                rows = result.fetchall()
                columns = result.keys()
                
                data = []
                for row in rows:
                    row_dict = {}
                    for j, col in enumerate(columns):
                        value = row[j]
                        row_dict[col] = value if value is None or isinstance(value, (int, float, str, bool)) else str(value)
                    data.append(row_dict)
                
                column_info = [{"name": col, "type": "unknown"} for col in columns]
                
                results.append(BatchSQLResult(
                    index=i,
                    sql=sql[:200] + "..." if len(sql) > 200 else sql,
                    sql_type=sql_type,
                    success=True,
                    message=f"查询成功，返回 {len(data)} 行",
                    row_count=len(data),
                    data=data,
                    columns=column_info,
                    execution_time=round(execution_time, 2)
                ))
            else:
                row_count = result.rowcount
                db.commit()
                
                results.append(BatchSQLResult(
                    index=i,
                    sql=sql[:200] + "..." if len(sql) > 200 else sql,
                    sql_type=sql_type,
                    success=True,
                    message=f"执行成功，影响 {row_count} 行",
                    row_count=row_count,
                    execution_time=round(execution_time, 2)
                ))
                
        except Exception as e:
            db.rollback()
            execution_time = (datetime.now() - start).total_seconds() * 1000
            
            results.append(BatchSQLResult(
                index=i,
                sql=sql[:200] + "..." if len(sql) > 200 else sql,
                sql_type=sql_type,
                success=False,
                message=f"执行失败: {str(e)}",
                execution_time=round(execution_time, 2),
                error=str(e)
            ))
    
    return results


async def _execute_in_batch(db: Session, statements: List[str]) -> List[BatchSQLResult]:
    """批处理模式 - 逐条执行，全部成功后统一提交"""
    results = []
    temp_results = []
    
    try:
        for i, sql in enumerate(statements):
            sql = sql.strip()
            if not sql:
                continue
                
            sql_type = get_sql_type(sql)
            start = datetime.now()
            
            result = db.execute(text(sql))
            execution_time = (datetime.now() - start).total_seconds() * 1000
            
            if sql_type == "SELECT" or sql_type == "WITH":
                rows = result.fetchall()
                columns = result.keys()
                
                data = []
                for row in rows:
                    row_dict = {}
                    for j, col in enumerate(columns):
                        value = row[j]
                        row_dict[col] = value if value is None or isinstance(value, (int, float, str, bool)) else str(value)
                    data.append(row_dict)
                
                column_info = [{"name": col, "type": "unknown"} for col in columns]
                
                temp_results.append({
                    "index": i,
                    "sql": sql[:200] + "..." if len(sql) > 200 else sql,
                    "sql_type": sql_type,
                    "success": True,
                    "message": f"查询成功，返回 {len(data)} 行",
                    "row_count": len(data),
                    "data": data,
                    "columns": column_info,
                    "execution_time": round(execution_time, 2)
                })
            else:
                row_count = result.rowcount
                temp_results.append({
                    "index": i,
                    "sql": sql[:200] + "..." if len(sql) > 200 else sql,
                    "sql_type": sql_type,
                    "success": True,
                    "message": f"执行成功，影响 {row_count} 行",
                    "row_count": row_count,
                    "execution_time": round(execution_time, 2)
                })
        
        # 全部成功，统一提交
        db.commit()
        
        for r in temp_results:
            results.append(BatchSQLResult(**r))
            
    except Exception as e:
        db.rollback()
        # 标记所有未完成的为失败
        for r in temp_results:
            results.append(BatchSQLResult(**r))
        
        # 添加错误信息
        results.append(BatchSQLResult(
            index=len(temp_results),
            sql="...",
            sql_type="OTHER",
            success=False,
            message=f"批处理失败，已回滚: {str(e)}",
            execution_time=0,
            error=str(e)
        ))
    
    return results


async def _execute_in_transaction(db: Session, statements: List[str]) -> List[BatchSQLResult]:
    """事务模式 - 任一失败则全部回滚"""
    results = []
    
    try:
        for i, sql in enumerate(statements):
            sql = sql.strip()
            if not sql:
                continue
                
            sql_type = get_sql_type(sql)
            start = datetime.now()
            
            result = db.execute(text(sql))
            execution_time = (datetime.now() - start).total_seconds() * 1000
            
            if sql_type == "SELECT" or sql_type == "WITH":
                rows = result.fetchall()
                columns = result.keys()
                
                data = []
                for row in rows:
                    row_dict = {}
                    for j, col in enumerate(columns):
                        value = row[j]
                        row_dict[col] = value if value is None or isinstance(value, (int, float, str, bool)) else str(value)
                    data.append(row_dict)
                
                column_info = [{"name": col, "type": "unknown"} for col in columns]
                
                results.append(BatchSQLResult(
                    index=i,
                    sql=sql[:200] + "..." if len(sql) > 200 else sql,
                    sql_type=sql_type,
                    success=True,
                    message=f"查询成功，返回 {len(data)} 行",
                    row_count=len(data),
                    data=data,
                    columns=column_info,
                    execution_time=round(execution_time, 2)
                ))
            else:
                row_count = result.rowcount
                results.append(BatchSQLResult(
                    index=i,
                    sql=sql[:200] + "..." if len(sql) > 200 else sql,
                    sql_type=sql_type,
                    success=True,
                    message=f"执行成功，影响 {row_count} 行",
                    row_count=row_count,
                    execution_time=round(execution_time, 2)
                ))
        
        # 全部成功，提交事务
        db.commit()
        
    except Exception as e:
        db.rollback()
        # 标记已执行的为成功，当前为失败
        for r in results:
            r.success = False
            r.message = f"事务回滚: {str(e)}"
        
        results.append(BatchSQLResult(
            index=len(results),
            sql=sql[:200] + "..." if len(sql) > 200 else sql,
            sql_type=get_sql_type(sql),
            success=False,
            message=f"事务执行失败，已回滚: {str(e)}",
            execution_time=0,
            error=str(e)
        ))
    
    return results
```

```vue
  modelValue: String,

  }
```

```vue
  get: () => props.modelValue,
  set: (val) => emit('update:modelValue', val)
})

const textarea = ref(null)
const showSuggestions = ref(false)
const activeIndex = ref(0)

const currentWord = ref('')
const dropdownStyle = ref({ top: '0px', left: '0px' })

  'SELECT', 'FROM', 'WHERE', 'INSERT', 'UPDATE', 'DELETE',
  'JOIN', 'LEFT', 'RIGHT', 'INNER', 'OUTER', 'ON',
  'GROUP BY', 'ORDER BY', 'HAVING', 'LIMIT', 'OFFSET',

  if (!currentWord.value) return []
  
  const word = currentWord.value.toLowerCase()
  return suggestions.value
    .filter(s => s.label.toLowerCase().includes(word))

}

  const el = textarea.value
  if (!el) return

  const text = el.value.substring(0, cursorPos)
  const lines = text.split('\n')
  const currentLine = lines[lines.length - 1]
  
  // 估算字符宽度和行高
  const charWidth = 8
  const lineHeight = 20
  
  const left = currentLine.length * charWidth
  const top = lines.length * lineHeight + 5
  
  dropdownStyle.value = {
    top: `${top}px`,
    left: `${left}px`
  }
}

// 输入处理
const onInput = () => {
  currentWord.value = getWordBeforeCursor()
  
  if (currentWord.value.length >= 1) {
    showSuggestions.value = true
    activeIndex.value = 0
    calculateDropdownPosition()
  } else {
    showSuggestions.value = false
  }
}

// 键盘事件
const onKeydown = (e) => {
  // Ctrl+Space 触发提示
  if (e.ctrlKey && e.code === 'Space') {
    e.preventDefault()
    currentWord.value = getWordBeforeCursor()
    showSuggestions.value = true
    activeIndex.value = 0
    calculateDropdownPosition()
    return
  }
  
  // Ctrl+Enter 执行
  if (e.ctrlKey && e.key === 'Enter') {
    e.preventDefault()
    emit('execute')
    return
  }
  
  // 提示框显示时的导航
  if (showSuggestions.value) {
    switch (e.key) {
      case 'ArrowDown':
        e.preventDefault()
        activeIndex.value = (activeIndex.value + 1) % filteredSuggestions.value.length
        break
      case 'ArrowUp':
        e.preventDefault()
        activeIndex.value = (activeIndex.value - 1 + filteredSuggestions.value.length) % filteredSuggestions.value.length
        break
      case 'Enter':
      case 'Tab':
        e.preventDefault()
        if (filteredSuggestions.value[activeIndex.value]) {
          selectSuggestion(filteredSuggestions.value[activeIndex.value])
        }
        break
      case 'Escape':
        showSuggestions.value = false
        break
    }
  }
}

// 选择提示项
const selectSuggestion = (item) => {
  const el = textarea.value
  if (!el) return
  
  const cursorPos = el.selectionStart
  const text = el.value
  const beforeCursor = text.substring(0, cursorPos)
  const afterCursor = text.substring(cursorPos)
  
  // 替换当前单词
  const word = getWordBeforeCursor()
  const newBeforeCursor = beforeCursor.substring(0, beforeCursor.length - word.length) + item.label
  
  sqlInput.value = newBeforeCursor + afterCursor
  showSuggestions.value = false
  
  // 设置光标位置
  setTimeout(() => {
    const newPos = newBeforeCursor.length
    el.setSelectionRange(newPos, newPos)
    el.focus()
  }, 0)
}

const hideSuggestions = () => {
  setTimeout(() => {
    showSuggestions.value = false
  }, 200)
}

const getTypeLabel = (type) => {
  const labels = {
    keyword: '关键字',
    table: '表',
    column: '列',
    function: '函数'
  }
  return labels[type] || type
}
```

## 九、注意事项

1. **安全性**：SQL执行接口需要添加权限验证，防止未授权访问
2. **SQL注入**：已添加基本的危险关键字过滤，生产环境建议更严格的验证

5. **事务**：批量执行时注意事务边界，避免长时间锁定表
