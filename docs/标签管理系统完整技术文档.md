## 一、系统概述

### 1.1 功能定位
标签管理系统用于统一管理股票的各类标签，包括：
- 题材标签（theme）
- 行业标签（industry）
- 概念标签（concept）
- 技术指标标签（technical）
- 地域标签（region）
- 风格标签（style）
- 基本面标签（fundamental）
- 自定义标签（custom）

### 1.2 核心特性
- 支持标签类型筛选和搜索
- 三栏式布局：标签列表 | 标签详情 | 关联股票
- 支持新建、编辑、删除标签
- 支持查看标签关联的股票列表
- 技术指标标签支持时间维度

### 3.3 服务层实现

```python
"""
股票标签服务 - 支持多种标签类型的统一管理
"""
from typing import List, Dict, Any, Optional
from datetime import date, datetime
from sqlalchemy.orm import Session
from sqlalchemy import and_, or_, func
from ..models.stock_tag import (
    StockTag, StockTagMapping, StockTechnicalTag, StockIndustry,
    StockTagHistory, TagType, TechnicalSignalType, IndustryStandard
)


class StockTagService:
    """股票标签服务类"""

    # ==================== 标签定义管理 ====================

    @staticmethod
    def create_tag(db: Session, tag_id: str, tag_name: str, tag_type: TagType, **kwargs) -> StockTag:
        """创建标签定义"""
        existing = db.query(StockTag).filter(StockTag.tag_id == tag_id).first()
        if existing:
            return existing

        tag = StockTag(
            tag_id=tag_id,
            tag_name=tag_name,
            tag_type=tag_type.value,
            tag_level=kwargs.get('tag_level', 1),
            parent_tag_id=kwargs.get('parent_tag_id'),
            keywords=kwargs.get('keywords'),
            description=kwargs.get('description'),
            color=kwargs.get('color'),
            icon=kwargs.get('icon'),
            source=kwargs.get('source'),
            source_code=kwargs.get('source_code')
        )
        db.add(tag)
        db.commit()
        db.refresh(tag)
        return tag

    @staticmethod
    def get_tag_by_id(db: Session, tag_id: str) -> Optional[StockTag]:
        """根据ID获取标签"""
        return db.query(StockTag).filter(StockTag.tag_id == tag_id).first()

    @staticmethod
    def get_tags_by_type(db: Session, tag_type: TagType, parent_tag_id: Optional[str] = None) -> List[StockTag]:
        """根据类型获取标签列表"""
        query = db.query(StockTag).filter(StockTag.tag_type == tag_type.value)
        if parent_tag_id is not None:
            query = query.filter(StockTag.parent_tag_id == parent_tag_id)
        return query.order_by(StockTag.stock_count.desc()).all()

    @staticmethod
    def search_tags(db: Session, keyword: str, tag_type: Optional[TagType] = None) -> List[StockTag]:
        """搜索标签"""
        query = db.query(StockTag).filter(
            or_(
                StockTag.tag_name.contains(keyword),
                StockTag.keywords.contains(keyword),
                StockTag.description.contains(keyword)
            )
        )
        if tag_type:
            query = query.filter(StockTag.tag_type == tag_type.value)
        return query.all()

    # ==================== 股票-标签关联管理 ====================

    @staticmethod
    def add_tag_to_stock(db: Session, code: str, market: str, tag_id: str,
                         weight: int = 0, operator: Optional[str] = None) -> StockTagMapping:
        """为股票添加标签"""
        tag = db.query(StockTag).filter(StockTag.tag_id == tag_id).first()
        if not tag:
            raise ValueError(f"标签不存在: {tag_id}")

        existing = db.query(StockTagMapping).filter(
            and_(StockTagMapping.code == code,
                 StockTagMapping.market == market,
                 StockTagMapping.tag_id == tag_id)
        ).first()

        if existing:
            return existing

        mapping = StockTagMapping(
            code=code, market=market, tag_id=tag_id,
            tag_type=tag.tag_type, weight=weight
        )
        db.add(mapping)
        StockTagService._record_history(db, code, market, tag_id, tag.tag_type, 'ADD', operator=operator)
        StockTagService._update_tag_stats(db, tag_id)
        db.commit()
        return mapping

    @staticmethod
    def get_stock_tags(db: Session, code: str, market: str,
                       tag_type: Optional[TagType] = None) -> List[Dict[str, Any]]:
        """获取股票的所有标签"""
        query = db.query(StockTagMapping, StockTag).join(
            StockTag, StockTagMapping.tag_id == StockTag.tag_id
        ).filter(and_(StockTagMapping.code == code, StockTagMapping.market == market))

        if tag_type:
            query = query.filter(StockTagMapping.tag_type == tag_type.value)

        results = query.all()
        return [{
            'tag_id': tag.tag_id,
            'tag_name': tag.tag_name,
            'tag_type': tag.tag_type,
            'color': tag.color,
            'weight': mapping.weight,
            'created_at': mapping.created_at
        } for mapping, tag in results]

    @staticmethod
    def get_stocks_by_tag(db: Session, tag_id: str, limit: Optional[int] = None) -> List[Dict[str, Any]]:
        """获取具有某标签的所有股票"""
        from ..models.stock import StockBasic

        query = db.query(StockTagMapping, StockBasic).outerjoin(
            StockBasic,
            and_(StockTagMapping.code == StockBasic.code,
                 StockTagMapping.market == StockBasic.market)
        ).filter(StockTagMapping.tag_id == tag_id)

        if limit:
            query = query.limit(limit)

        results = query.all()
        return [{
            'code': mapping.code,
            'market': mapping.market,
            'name': basic.name if basic else '',
            'industry': basic.industry if basic else '',
            'weight': mapping.weight
        } for mapping, basic in results]

    # ==================== 技术指标标签管理 ====================

    @staticmethod
    def add_technical_tag(db: Session, code: str, market: str, indicator_name: str,
                          trade_date: date, signal_type: Optional[str] = None, **kwargs) -> StockTechnicalTag:
        """添加技术指标标签"""
        existing = db.query(StockTechnicalTag).filter(
            and_(StockTechnicalTag.code == code,
                 StockTechnicalTag.market == market,
                 StockTechnicalTag.indicator_name == indicator_name,
                 StockTechnicalTag.trade_date == trade_date,
                 StockTechnicalTag.signal_type == signal_type)
        ).first()

        if existing:
            return existing

        tech_tag = StockTechnicalTag(
            code=code, market=market, indicator_name=indicator_name,
            signal_type=signal_type, trade_date=trade_date,
            trade_time=kwargs.get('trade_time'), freq=kwargs.get('freq', 'day'),
            indicator_value=kwargs.get('indicator_value'),
            signal_strength=kwargs.get('signal_strength', 3),
            trigger_price=kwargs.get('trigger_price'),
            trigger_volume=kwargs.get('trigger_volume'),
            description=kwargs.get('description'),
            is_active=kwargs.get('is_active', 1),
            confirmed=kwargs.get('confirmed', 0)
        )
        db.add(tech_tag)
        db.commit()
        db.refresh(tech_tag)
        return tech_tag

    @staticmethod
    def get_technical_tags(db: Session, code: Optional[str] = None, market: Optional[str] = None,
                           indicator_name: Optional[str] = None, signal_type: Optional[str] = None,
                           trade_date: Optional[date] = None, start_date: Optional[date] = None,
                           end_date: Optional[date] = None, is_active: Optional[int] = None,
                           limit: int = 100) -> List[StockTechnicalTag]:
        """查询技术指标标签（支持多种过滤条件）"""
        query = db.query(StockTechnicalTag)

        if code:
            query = query.filter(StockTechnicalTag.code == code)
        if market:
            query = query.filter(StockTechnicalTag.market == market)
        if indicator_name:
            query = query.filter(StockTechnicalTag.indicator_name == indicator_name)
        if signal_type:
            query = query.filter(StockTechnicalTag.signal_type == signal_type)
        if trade_date:
            query = query.filter(StockTechnicalTag.trade_date == trade_date)
        if start_date:
            query = query.filter(StockTechnicalTag.trade_date >= start_date)
        if end_date:
            query = query.filter(StockTechnicalTag.trade_date <= end_date)
        if is_active is not None:
            query = query.filter(StockTechnicalTag.is_active == is_active)

        return query.order_by(StockTechnicalTag.trade_date.desc()).limit(limit).all()

    @staticmethod
    def get_stocks_by_technical_signal(db: Session, indicator_name: str, signal_type: str,
                                       trade_date: date, min_strength: int = 1) -> List[Dict[str, Any]]:
        """获取某日在某指标上发出特定信号的所有股票"""
        from ..models.stock import StockBasic

        results = db.query(StockTechnicalTag, StockBasic).outerjoin(
            StockBasic,
            and_(StockTechnicalTag.code == StockBasic.code,
                 StockTechnicalTag.market == StockBasic.market)
        ).filter(
            and_(StockTechnicalTag.indicator_name == indicator_name,
                 StockTechnicalTag.signal_type == signal_type,
                 StockTechnicalTag.trade_date == trade_date,
                 StockTechnicalTag.signal_strength >= min_strength,
                 StockTechnicalTag.is_active == 1)
        ).order_by(StockTechnicalTag.signal_strength.desc()).all()

        return [{
            'code': tag.code, 'market': tag.market,
            'name': basic.name if basic else '',
            'indicator_value': tag.indicator_value,
            'signal_strength': tag.signal_strength,
            'trigger_price': tag.trigger_price,
            'description': tag.description
        } for tag, basic in results]

    # ==================== 行业分类管理 ====================

    @staticmethod
    def set_stock_industry(db: Session, code: str, market: str, standard: IndustryStandard,
                           level1_code: str, level1_name: str, **kwargs) -> StockIndustry:
        """设置股票行业分类"""
        industry = db.query(StockIndustry).filter(
            and_(StockIndustry.code == code,
                 StockIndustry.market == market,
                 StockIndustry.standard == standard.value)
        ).first()

        if industry:
            industry.industry_level1_code = level1_code
            industry.industry_level1_name = level1_name
            industry.industry_level2_code = kwargs.get('level2_code')
            industry.industry_level2_name = kwargs.get('level2_name')
            industry.industry_level3_code = kwargs.get('level3_code')
            industry.industry_level3_name = kwargs.get('level3_name')
            industry.effective_date = kwargs.get('effective_date') or date.today()
        else:
            industry = StockIndustry(
                code=code, market=market, standard=standard.value,
                industry_level1_code=level1_code, industry_level1_name=level1_name,
                industry_level2_code=kwargs.get('level2_code'),
                industry_level2_name=kwargs.get('level2_name'),
                industry_level3_code=kwargs.get('level3_code'),
                industry_level3_name=kwargs.get('level3_name'),
                effective_date=kwargs.get('effective_date') or date.today()
            )
            db.add(industry)

        db.commit()
        db.refresh(industry)
        return industry

    # ==================== 私有辅助方法 ====================

    @staticmethod
    def _record_history(db: Session, code: str, market: str, tag_id: str,

        """记录标签操作历史"""
        history = StockTagHistory(

            action=action, old_value=kwargs.get('old_value'),
            new_value=kwargs.get('new_value'), operator=kwargs.get('operator'),
            source=kwargs.get('source', 'api')
        )
        db.add(history)

    @staticmethod
    def _update_tag_stats(db: Session, tag_id: str):
        """更新标签统计信息"""
        count = db.query(StockTagMapping).filter(StockTagMapping.tag_id == tag_id).count()
        tag = db.query(StockTag).filter(StockTag.tag_id == tag_id).first()

<script setup>
defineOptions({ name: 'StockTag' })
import { ref, computed, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import { ElMessage, ElMessageBox } from 'element-plus'
import {
  PriceTag, CollectionTag, OfficeBuilding, Star,
  TrendCharts, MapLocation, Search, RefreshRight,
  Plus, List, InfoFilled, Edit, Delete, User
} from '@element-plus/icons-vue'
import axios from 'axios'

const router = useRouter()
const API_BASE_URL = import.meta.env.VITE_API_BASE_URL || 'http://localhost:4000'

// ==================== 状态变量 ====================
const loading = ref(false)
const stockLoading = ref(false)
const submitLoading = ref(false)
const searchKeyword = ref('')
const currentTagType = ref('')
const tagList = ref([])
const selectedTag = ref(null)
const tagStocks = ref([])
const dialogVisible = ref(false)
const isEdit = ref(false)
const tagFormRef = ref(null)

const tagForm = ref({
  tag_id: '',
  tag_name: '',
  tag_type: 'theme',
  keywords: '',
  description: '',
  color: '#409EFF'
})

const tagRules = {
  tag_name: [{ required: true, message: '请输入标签名称', trigger: 'blur' }],
  tag_type: [{ required: true, message: '请选择标签类型', trigger: 'change' }],
  tag_id: [{ required: true, message: '请输入标签ID', trigger: 'blur' }]
}

// 标签类型映射
const tagTypeMap = {
  theme: { label: '题材', color: 'primary' },
  industry: { label: '行业', color: 'success' },
  concept: { label: '概念', color: 'warning' },
  technical: { label: '技术指标', color: 'danger' },
  region: { label: '地域', color: 'info' },
  style: { label: '风格', color: '' }
}

// ==================== 计算属性 ====================
const currentTagTypeLabel = computed(() => {
  if (!currentTagType.value) return '全部标签'
  return tagTypeMap[currentTagType.value]?.label + '标签' || '标签列表'
})

const filteredTags = computed(() => {
  let result = tagList.value
  if (currentTagType.value) {
    result = result.filter(tag => tag.tag_type === currentTagType.value)
  }
  if (searchKeyword.value) {
    const keyword = searchKeyword.value.toLowerCase()
    result = result.filter(tag =>
      tag.tag_name.toLowerCase().includes(keyword) ||
      (tag.keywords && tag.keywords.toLowerCase().includes(keyword))
    )
  }
  return result
})

const tagStats = computed(() => ({ total: tagList.value.length }))

// ==================== 方法 ====================
const getTagTypeLabel = (type) => tagTypeMap[type]?.label || type
const getTagTypeColor = (type) => tagTypeMap[type]?.color || ''
const formatDate = (dateStr) => {
  if (!dateStr) return '-'
  return new Date(dateStr).toLocaleString('zh-CN')
}

// 获取标签列表
const fetchTagList = async () => {
  loading.value = true
  try {
    const response = await axios.get(`${API_BASE_URL}/api/stock-tags/tags`)
    tagList.value = response.data || []
  } catch (error) {
    console.error('获取标签列表失败:', error)
    ElMessage.error('获取标签列表失败')
  } finally {
    loading.value = false
  }
}

// 搜索标签
const handleSearch = async () => {
  if (!searchKeyword.value) {
    await fetchTagList()
    return
  }
  loading.value = true
  try {
    const response = await axios.get(`${API_BASE_URL}/api/stock-tags/tags/search`, {
      params: { keyword: searchKeyword.value }
    })
    tagList.value = response.data || []
  } catch (error) {
    console.error('搜索标签失败:', error)
    ElMessage.error('搜索失败')
  } finally {
    loading.value = false
  }
}

// 重置搜索
const resetSearch = () => {
  searchKeyword.value = ''
  currentTagType.value = ''
  fetchTagList()
}

// 切换标签类型
const handleTagTypeChange = () => {
  selectedTag.value = null
  tagStocks.value = []
}

// 选择标签
const selectTag = async (tag) => {
  selectedTag.value = tag
  await fetchTagStocks(tag.tag_id)
}

// 获取标签关联的股票
const fetchTagStocks = async (tagId) => {
  stockLoading.value = true
  try {
    const response = await axios.get(`${API_BASE_URL}/api/stock-tags/tag/${tagId}/stocks`)
    tagStocks.value = response.data?.data || []
  } catch (error) {
    console.error('获取标签股票失败:', error)
    ElMessage.error('获取关联股票失败')
  } finally {
    stockLoading.value = false
  }
}

// 显示创建对话框
const showCreateDialog = () => {
  isEdit.value = false
  tagForm.value = {
    tag_id: '',
    tag_name: '',
    tag_type: currentTagType.value || 'theme',
    keywords: '',
    description: '',
    color: '#409EFF'
  }
  dialogVisible.value = true
}

// 显示编辑对话框
const showEditDialog = () => {
  if (!selectedTag.value) return
  isEdit.value = true
  tagForm.value = { ...selectedTag.value }
  dialogVisible.value = true
}

// 提交表单
const handleSubmit = async () => {
  const valid = await tagFormRef.value.validate().catch(() => false)
  if (!valid) return

  submitLoading.value = true
  try {
    if (isEdit.value) {
      ElMessage.success('标签更新成功')
    } else {
      await axios.post(`${API_BASE_URL}/api/stock-tags/tags`, tagForm.value)
      ElMessage.success('标签创建成功')
    }
    dialogVisible.value = false
    await fetchTagList()
  } catch (error) {
    console.error('保存标签失败:', error)
    ElMessage.error(error.response?.data?.detail || '保存失败')
  } finally {
    submitLoading.value = false
  }
}

// 删除标签
const handleDeleteTag = async () => {
  if (!selectedTag.value) return
  try {
    await ElMessageBox.confirm(
      `确定要删除标签 "${selectedTag.value.tag_name}" 吗？`,
      '确认删除',
      { type: 'warning' }
    )
    ElMessage.success('标签删除成功')
    selectedTag.value = null
    await fetchTagList()
  } catch (error) {
    if (error !== 'cancel') {
      console.error('删除标签失败:', error)
      ElMessage.error('删除失败')
    }
  }
}

// 跳转到股票详情
const goToStockDetail = (stock) => {
  router.push({
    path: '/stock',
    query: { code: stock.code, market: stock.market }
  })
}

// 初始化
onMounted(() => {
  fetchTagList()
})
</script>

## 五、交互逻辑详细说明

### 5.1 页面加载流程

```
┌─────────────────────────────────────────────────────────────┐
│  1. 页面初始化                                               │
│     └─> onMounted() 调用 fetchTagList()                     │
│         └─> GET /api/stock-tags/tags                        │
│             └─> 加载全部标签列表                             │
├─────────────────────────────────────────────────────────────┤
│  2. 用户选择标签类型                                         │
│     └─> handleTagTypeChange()                               │
│         └─> 清空选中状态和股票列表                           │
│         └─> filteredTags 计算属性自动过滤                    │
├─────────────────────────────────────────────────────────────┤
│  3. 用户点击标签                                             │
│     └─> selectTag(tag)                                      │
│         └─> 更新 selectedTag                                │
│         └─> 调用 fetchTagStocks(tag.tag_id)                 │
│             └─> GET /api/stock-tags/tag/{tag_id}/stocks     │
│                 └─> 加载关联股票列表                         │
└─────────────────────────────────────────────────────────────┘
```

### 5.2 搜索功能流程

```
用户输入关键词 ──> handleSearch()
    │
    ├─> 有关键词？
    │   ├─> 是：GET /api/stock-tags/tags/search?keyword=xxx
    │   └─> 否：GET /api/stock-tags/tags
    │
    └─> 更新 tagList
```

### 5.3 新建标签流程

```
点击"新建标签" ──> showCreateDialog()
    │
    ├─> isEdit = false
    ├─> 重置 tagForm
    └─> dialogVisible = true
