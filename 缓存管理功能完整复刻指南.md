## 一、功能概述

缓存管理功能是一个完整的缓存监控和管理系统，包含：
- **内存监控**：实时显示系统内存和进程内存使用情况
- **缓存统计**：显示缓存总数、有效缓存、过期缓存数量
- **缓存列表**：分页展示所有缓存记录，支持过滤（全部/有效/过期）
- **缓存详情**：查看单个缓存的详细信息和内容
- **缓存操作**：删除单个缓存、清除过期缓存、清除所有缓存

```python
"""
缓存键生成工具
提供统一的缓存键生成逻辑，确保预热服务、CacheService和同步API使用一致的缓存键
"""
import hashlib
import json
from typing import Dict, Any


class CacheKeyGenerator:
    """缓存键生成器"""
    
    @staticmethod
    def generate_stock_list_key(market: str = None) -> str:
        """
        生成股票列表缓存键
        
        Args:
            market: 市场代码(SH/SZ)，None表示全部市场
        
        Returns:
            缓存键，格式: stock_list:{market}
        """
        market_key = market.upper() if market else "all"
        return f"stock_list:{market_key}"
    
    @staticmethod
    def generate_stock_quote_key(code: str, market: str) -> str:
        """
        生成股票实时行情缓存键
        
        Args:
            code: 股票代码
            market: 市场代码(SH/SZ)
        
        Returns:
            缓存键，格式: stock_quote:{market}:{code}
        """
        return f"stock_quote:{market.upper()}:{code}"
    
    @staticmethod
    def generate_kline_key(code: str, market: str, freq: str = "day", 
                         start: str = None, end: str = None) -> str:
        """
        生成K线数据缓存键
        
        Args:
            code: 股票代码
            market: 市场代码(SH/SZ)
            freq: 周期
            start: 开始日期
            end: 结束日期
        
        Returns:
            缓存键，格式: kline:{market}:{code}:{freq}:{start}:{end}
        """
        params = {
            "code": code,
            "market": market.upper(),
            "freq": freq,
            "start": start or "",
            "end": end or ""
        }
        sorted_params = sorted(params.items())
        param_str = json.dumps(sorted_params, sort_keys=True)
        hash_value = hashlib.md5(param_str.encode()).hexdigest()
        return f"kline:{hash_value}"
    
    @staticmethod
    def generate_transaction_key(code: str, market: str, date: str = None) -> str:
        """
        生成分笔数据缓存键
        
        Args:
            code: 股票代码
            market: 市场代码(SH/SZ)
            date: 日期
        
        Returns:
            缓存键，格式: transaction:{market}:{code}:{date}
        """
        params = {
            "code": code,
            "market": market.upper(),
            "date": date or ""
        }
        sorted_params = sorted(params.items())
        param_str = json.dumps(sorted_params, sort_keys=True)
        hash_value = hashlib.md5(param_str.encode()).hexdigest()
        return f"transaction:{hash_value}"
    
    @staticmethod
    def generate_index_key(code: str, market: str) -> str:
        """
        生成指数行情缓存键
        
        Args:
            code: 指数代码
            market: 市场代码(SH/SZ)
        
        Returns:
            缓存键，格式: index:{market}:{code}
        """
        return f"index:{market.upper()}:{code}"
    
    @staticmethod
    def generate_generic_key(prefix: str, **params) -> str:
        """
        生成通用缓存键（使用MD5哈希）
        
        Args:
            prefix: 缓存键前缀
            **params: 缓存参数
        
        Returns:
            缓存键，格式: {prefix}:{md5_hash}
        """
        sorted_params = sorted(params.items())
        param_str = json.dumps(sorted_params, sort_keys=True)
        hash_value = hashlib.md5(param_str.encode()).hexdigest()
        return f"{prefix}:{hash_value}"
```

```python
"""
数据缓存服务
处理数据库缓存和接口调用的协调
"""
from typing import Optional, List, Dict, Any, Callable
from datetime import datetime, timedelta
from sqlalchemy.orm import Session
from sqlalchemy import and_
import pandas as pd
import hashlib
import json

from ..models.stock import StockBasic, StockQuote, StockKline, StockTick, CacheRecord
from ..core.config import settings
from ..core.logger import setup_logging
from .tdx_service import tdx_service
from ..utils.cache_key_generator import cache_key_generator

logger = setup_logging()


class CacheService:
    """缓存服务"""
    
    def __init__(self, db: Session):
        self.db = db
    
    def _generate_cache_key(self, prefix: str, **params) -> str:
        """生成缓存键（使用统一的缓存键生成器）"""
        # 股票列表使用简单字符串格式
        if prefix == "stock_list":
            market = params.get("market", "all")
            return cache_key_generator.generate_stock_list_key(market)
        
        # 股票实时行情使用简单字符串格式
        if prefix == "stock_quote":
            code = params.get("code", "")
            market = params.get("market", "")
            return cache_key_generator.generate_stock_quote_key(code, market)
        
        # 指数行情使用简单字符串格式
        if prefix == "index":
            code = params.get("code", "")
            market = params.get("market", "")
            return cache_key_generator.generate_index_key(code, market)
        
        # 其他数据使用哈希格式
        return cache_key_generator.generate_generic_key(prefix, **params)
    
    def _generate_data_hash(self, data: Any) -> str:
        """生成数据哈希"""
        data_str = json.dumps(data, sort_keys=True, default=str)
        return hashlib.sha256(data_str.encode()).hexdigest()
    
    def _is_cache_valid(self, cache_key: str) -> Optional[CacheRecord]:
        """检查缓存是否有效"""
        cache = self.db.query(CacheRecord).filter(
            CacheRecord.cache_key == cache_key
        ).first()
        
        if cache and cache.expire_time > datetime.now():
            return cache
        return None
    
    def _update_cache(self, cache_key: str, data_hash: str, data: Any = None, cache_type: str = "default"):
        """更新缓存记录
        
        Args:
            cache_key: 缓存键
            data_hash: 数据哈希
            data: 缓存数据(将序列化为JSON)
            cache_type: 缓存类型 (stock_list/index/quote/default)
        """
        # 根据缓存类型设置不同的过期时间
        if cache_type == "stock_list":
            expire_time = datetime.now() + timedelta(hours=settings.STOCK_LIST_CACHE_HOURS)
        elif cache_type == "index":
            expire_time = datetime.now() + timedelta(minutes=settings.INDEX_CACHE_MINUTES)
        elif cache_type == "quote":
            expire_time = datetime.now() + timedelta(seconds=settings.QUOTE_CACHE_SECONDS)
        elif cache_type == "kline":
            # K线数据缓存7天，历史数据不常变化
            expire_time = datetime.now() + timedelta(days=settings.KLINE_CACHE_DAYS)
        else:
            expire_time = datetime.now() + timedelta(minutes=settings.CACHE_EXPIRE_MINUTES)
        
        # 序列化缓存数据
        cache_data_str = None
        if data is not None:
            try:
                if isinstance(data, pd.DataFrame):
                    cache_data_str = data.to_json(force_ascii=False, orient='records')
                else:
                    cache_data_str = json.dumps(data, ensure_ascii=False, default=str)
            except Exception as e:
                logger.warning(f"[缓存] 序列化缓存数据失败: {e}")
        
        cache = self.db.query(CacheRecord).filter(
            CacheRecord.cache_key == cache_key
        ).first()
        
        if cache:
            cache.data_hash = data_hash
            cache.cache_data = cache_data_str
            cache.expire_time = expire_time
        else:
            cache = CacheRecord(
                cache_key=cache_key,
                data_hash=data_hash,
                cache_data=cache_data_str,
                expire_time=expire_time
            )
            self.db.add(cache)
        
        self.db.commit()
```

```python
def get_memory_info() -> Dict[str, Any]:
    """
    获取内存使用信息
    
    Returns:
        内存使用信息字典
    """
    try:
        import psutil
        
        # 获取系统内存信息
        mem = psutil.virtual_memory()
        
        # 获取当前进程内存使用
        process = psutil.Process()
        process_mem = process.memory_info()
        
        return {
            "total": mem.total,
            "total_formatted": format_size(mem.total),
            "available": mem.available,
            "available_formatted": format_size(mem.available),
            "used": mem.used,
            "used_formatted": format_size(mem.used),
            "percent": mem.percent,
            "process_rss": process_mem.rss,
            "process_rss_formatted": format_size(process_mem.rss),
            "process_vms": process_mem.vms,
            "process_vms_formatted": format_size(process_mem.vms)
        }
    except ImportError:
        logger.warning("[内存信息] psutil模块未安装，无法获取内存信息")
        return {
            "total": 0,
            "total_formatted": "N/A",
            "available": 0,

def format_size(size_bytes: int) -> str:
    """
    将字节数格式化为可读字符串
    
    Args:
        size_bytes: 字节数
        
    Returns:
        格式化后的字符串
    """
    if size_bytes < 1024:
        return f"{size_bytes} B"
    elif size_bytes < 1024 * 1024:
        return f"{size_bytes / 1024:.2f} KB"
    elif size_bytes < 1024 * 1024 * 1024:
        return f"{size_bytes / (1024 * 1024):.2f} MB"
    else:
        return f"{size_bytes / (1024 * 1024 * 1024):.2f} GB"


@router.get("/list", summary="获取缓存列表")
def get_cache_list(
    page: int = Query(1, ge=1, description="页码"),
    page_size: int = Query(20, ge=1, le=100, description="每页数量"),
    filter_type: str = Query("", description="过滤类型: valid=有效, expired=过期, 空=全部"),
    db: Session = Depends(get_db)
) -> Dict[str, Any]:
    """
    获取缓存列表（分页）
    
    Args:
        page: 页码
        page_size: 每页数量
        filter_type: 过滤类型
        db: 数据库会话
        
    Returns:
        缓存列表数据
    """
    logger.info(f"[方法入口] get_cache_list - page={page}, page_size={page_size}, filter_type={filter_type}")
    
    try:
        # 构建查询条件
        query = db.query(CacheRecord)
        
        # 应用过滤条件
        if filter_type == "valid":
            query = query.filter(CacheRecord.expire_time >= datetime.now())
        elif filter_type == "expired":
            query = query.filter(CacheRecord.expire_time < datetime.now())
        
        # 查询总记录数
        total = query.with_entities(func.count(CacheRecord.id)).scalar()
        
        # 查询分页数据
        offset = (page - 1) * page_size
        cache_records = query.order_by(
            CacheRecord.created_at.desc()
        ).offset(offset).limit(page_size).all()
        
        # 获取内存信息
        memory_info = get_memory_info()
        
        # 构建返回数据
        data = []
        for record in cache_records:
            is_expired = record.expire_time < datetime.now()
            data.append({
                "id": record.id,
                "cache_key": record.cache_key,
                "data_hash": record.data_hash,
                "cache_data": record.cache_data,
                "expire_time": record.expire_time.isoformat(),
                "is_expired": is_expired,
                "created_at": record.created_at.isoformat(),
                "updated_at": record.updated_at.isoformat()
            })
        
        result = {
            "success": True,
            "data": {
                "list": data,
                "pagination": {
                    "page": page,
                    "page_size": page_size,
                    "total": total,
                    "total_pages": (total + page_size - 1) // page_size
                },
                "memory": memory_info
            }
        }
        
        logger.info(f"[方法出口] get_cache_list - 成功, 返回缓存数量: {len(data)}")
        return result
        
    except Exception as e:
        logger.error(f"[方法出口] get_cache_list - 失败, 错误: {str(e)}")
        return {
            "success": False,
            "message": f"获取缓存列表失败: {str(e)}",
            "data": None
        }

}
```

```vue
const formatRemainingTime = (expireTimeStr, isExpired) => {
  if (isExpired) {
    return '已过期'
  }
  if (!expireTimeStr) return '-'
  const expireTime = new Date(expireTimeStr)
  const now = new Date()
  const diff = expireTime - now
  
  if (diff <= 0) {
    return '已过期'
  }
  
  const days = Math.floor(diff / (1000 * 60 * 60 * 24))
  const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60))
  const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60))
  
  if (days > 0) {
    return `${days}天${hours}小时`
  } else if (hours > 0) {
    return `${hours}小时${minutes}分钟`
  } else {
    return `${minutes}分钟`
  }
}

// 方法：格式化缓存数据
const formatCacheData = (cacheData) => {
  if (!cacheData) return '-'
  try {
    const parsed = JSON.parse(cacheData)
    return JSON.stringify(parsed, null, 2)
  } catch (e) {
    return cacheData
  }
}
```

```python
def format_size(size_bytes: int) -> str:
    if size_bytes < 1024:
        return f"{size_bytes} B"
    elif size_bytes < 1024 * 1024:
        return f"{size_bytes / 1024:.2f} KB"
    elif size_bytes < 1024 * 1024 * 1024:
        return f"{size_bytes / (1024 * 1024):.2f} MB"
    else:
        return f"{size_bytes / (1024 * 1024 * 1024):.2f} GB"
```
